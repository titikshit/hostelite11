🗄️ DATABASE SETUP (SUPABASE):
================================================================================

1. CREATE TABLES:
```sql
-- Profiles table
CREATE TABLE profiles (
  id UUID REFERENCES auth.users(id) PRIMARY KEY,
  full_name TEXT,
  phone TEXT,
  role TEXT NOT NULL CHECK (role IN ('student', 'admin')),
  hostel_id TEXT NOT NULL,
  room_no TEXT,
  course TEXT,
  year TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Notices table
CREATE TABLE notices (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  hostel_id TEXT NOT NULL,
  title TEXT NOT NULL,
  body TEXT,
  tag TEXT DEFAULT 'General',
  created_by UUID REFERENCES auth.users(id),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Complaints table
CREATE TABLE complaints (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  hostel_id TEXT NOT NULL,
  student_id UUID REFERENCES auth.users(id),
  room_no TEXT,
  title TEXT,
  description TEXT NOT NULL,
  severity TEXT DEFAULT 'Low',
  status TEXT DEFAULT 'open',
  attachment_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Fees table
CREATE TABLE fees (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  student_id UUID REFERENCES auth.users(id),
  hostel_id TEXT NOT NULL,
  amount DECIMAL(10,2) NOT NULL,
  due_date DATE,
  status TEXT DEFAULT 'pending',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Rooms table
CREATE TABLE rooms (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  hostel_id TEXT NOT NULL,
  room_no TEXT NOT NULL,
  capacity INTEGER DEFAULT 2,
  occupied INTEGER DEFAULT 0,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

2. ENABLE ROW LEVEL SECURITY:
```sql
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE notices ENABLE ROW LEVEL SECURITY;
ALTER TABLE complaints ENABLE ROW LEVEL SECURITY;
ALTER TABLE fees ENABLE ROW LEVEL SECURITY;
ALTER TABLE rooms ENABLE ROW LEVEL SECURITY;
```

3. CREATE SIMPLIFIED RLS POLICIES:
```sql
-- Notices policies
CREATE POLICY "notices_select_simple" ON notices FOR SELECT USING (true);
CREATE POLICY "notices_insert_simple" ON notices FOR INSERT WITH CHECK (auth.uid() IS NOT NULL);
CREATE POLICY "notices_update_simple" ON notices FOR UPDATE USING (auth.uid() IS NOT NULL);

-- Profiles policies
CREATE POLICY "profiles_select_simple" ON profiles FOR SELECT USING (true);
CREATE POLICY "profiles_insert_simple" ON profiles FOR INSERT WITH CHECK (auth.uid() IS NOT NULL);
CREATE POLICY "profiles_update_simple" ON profiles FOR UPDATE USING (auth.uid() = id OR auth.uid() IN (SELECT id FROM profiles WHERE role = 'admin'));

-- Complaints policies
CREATE POLICY "complaints_select_simple" ON complaints FOR SELECT USING (true);
CREATE POLICY "complaints_insert_simple" ON complaints FOR INSERT WITH CHECK (auth.uid() IS NOT NULL);
CREATE POLICY "complaints_update_simple" ON complaints FOR UPDATE USING (auth.uid() IS NOT NULL);

-- Fees policies
CREATE POLICY "fees_select_simple" ON fees FOR SELECT USING (true);
CREATE POLICY "fees_insert_simple" ON fees FOR INSERT WITH CHECK (auth.uid() IS NOT NULL);
```

🔧 IMPORTANT DEBUGGING FIXES:
================================================================================

1. STACK DEPTH ERROR FIX:
   Problem: PostgreSQL stack depth limit exceeded when creating notices
   Solution: Simplified RLS policies (see RLS section above)

2. FLUTTER ANALYSIS ERRORS:
   - Use .withValues(alpha: 0.2) instead of .withOpacity(0.2)
   - Use const constructors where possible
   - Remove unused imports

3. DROPDOWN FORM FIELD DEPRECATION:
   Replace DropdownButtonFormField with:
   ```dart
   InputDecorator(
     decoration: InputDecoration(...),
     child: DropdownButtonHideUnderline(
       child: DropdownButton<String>(...),
     ),
   )
   ```

4. HOSTEL ID MISMATCH:
   Ensure student and admin are in same hostel:
   ```sql
   UPDATE profiles SET hostel_id = 'bose_hostel' 
   WHERE id = (SELECT id FROM auth.users WHERE email = 'student@test.com');
   ```
